% 1) Redefine the header to use \smallskip, not \medskip
%    and to drop the final \medskip entirely.
\renewcommand{\makecvheader}{%
  \begingroup
    \altacv@left@photos\hfill%
    \begin{minipage}{\dimexpr\linewidth-\altacv@photos@width}%
      \raggedright%
      {\namefont\color{name}\MakeUppercase{\@name}\par}%
      \smallskip% was \medskip in v1.7.x :contentReference[oaicite:3]{index=3}
      {\taglinefont\color{tagline}\@tagline\par}%
      \smallskip% tighten here too
      {\personalinfofont\@personalinfo\par}%
    \end{minipage}\hfill%
    \altacv@right@photos\par%
  \endgroup% no final \medskip
}

% 2) Restore v1.1.5 section‑heading spacing (medskip, not bigskip)
\renewcommand{\cvsection}[2][]{%
  \nointerlineskip
  \medskip% was \bigskip in v1.7.x :contentReference[oaicite:4]{index=4}
  \ifstrequal{#1}{}{}{\marginpar{%
    \vspace*{\dimexpr1pt-\baselineskip}\raggedright\input{#1}}}%
  {\color{heading}\cvsectionfont\MakeUppercase{#2}}\\[-1ex]%
  {\color{headingrule}\rule{\linewidth}{2pt}\par}\medskip
}

% —— Override \cvevent to the v1.1.5 version, with adjustable spacing
\renewcommand{\cvevent}[4]{%
  \noindent
    {\large\color{emphasis}#1}%   % job title
  \hfill
    \ifstrequal{#2}{}{}{\textbf{\color{accent}#2}}%   % company
  \\[-0.5ex]%                       % small negative gap before date line
    \ifstrequal{#3}{}{}{{\small\faCalendar\hspace{0.5em}#3}}%
  \hfill
    \ifstrequal{#4}{}{}{{\small\faMapMarker\hspace{0.5em}#4}}%
  \vspace{0.5ex}%                  % small gap after the block
}
