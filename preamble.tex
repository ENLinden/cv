\renewcommand{\cvsection}[2][]{%
  \par\bigskip
  \if\relax\detokenize{#1}\relax
  \else
    \marginpar{\raggedright\scriptsize\itshape #1\par}%
  \fi
  \noindent
    {\RaggedRight\color{heading}\cvsectionfont\MakeUppercase{#2}\par}%
  \noindent{\color{headingrule}\rule{\linewidth}{2pt}}\par\medskip
}
